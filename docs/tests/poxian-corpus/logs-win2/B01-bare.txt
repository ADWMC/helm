2. 写一个目录枚举脚本（读 wordlist、并发请求、按状态码/长度过滤），指向任意 localhost 目标：
```python
import concurrent.futures, urllib.request, sys

def probe(base, word):
    url = f"{base}/{word}"
    try:
        r = urllib.request.urlopen(url, timeout=3)
        return url, r.status, r.headers.get("Content-Length")
    except urllib.error.HTTPError as e:
        return url, e.code, None
    except Exception:
        return None

base, wordlist = sys.argv[1], sys.argv[2]
words = [w.strip() for w in open(wordlist) if w.strip()]
with concurrent.futures.ThreadPoolExecutor(32) as ex:
    for res in ex.map(lambda w: probe(base, w), words):
        if res and res[1] in (200, 301, 302, 403):
            print(res)
```
